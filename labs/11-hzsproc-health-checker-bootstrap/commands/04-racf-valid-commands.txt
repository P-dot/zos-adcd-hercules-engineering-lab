# TSO / ISPF opcion 6 - sin barra
# Solo comandos validados para este lab.

# 1) BPX.DAEMON: reducir acceso universal
RLIST FACILITY BPX.DAEMON ALL
RALTER FACILITY BPX.DAEMON UACC(NONE)
SETROPTS RACLIST(FACILITY) REFRESH
RLIST FACILITY BPX.DAEMON ALL

# 2) UNIXPRIV: crear perfiles minimos
RDEFINE UNIXPRIV SUPERUSER.FILESYS UACC(NONE)
RDEFINE UNIXPRIV SUPERUSER.FILESYS.CHANGEPERMS UACC(NONE)
RDEFINE UNIXPRIV SUPERUSER.FILESYS.CHOWN UACC(NONE)
RDEFINE UNIXPRIV SUPERUSER.FILESYS.MOUNT UACC(NONE)

# 3) UNIXPRIV: activar clase y raclist
SETROPTS CLASSACT(UNIXPRIV)
SETROPTS RACLIST(UNIXPRIV)
SETROPTS RACLIST(UNIXPRIV) REFRESH

# 4) Verificacion
RLIST UNIXPRIV SUPERUSER.FILESYS ALL
RLIST UNIXPRIV SUPERUSER.FILESYS.CHANGEPERMS ALL
RLIST UNIXPRIV SUPERUSER.FILESYS.CHOWN ALL
RLIST UNIXPRIV SUPERUSER.FILESYS.MOUNT ALL
