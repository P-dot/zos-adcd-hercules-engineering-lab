# Lab 32 - local installation and Git publication
# Run from the repository root.

ZIP="$HOME/Downloads/lab-32-multistep-jcl-return-codes-conditional-execution.zip"
TMP="$HOME/Downloads/lab32-tmp"
LAB="labs/32-multistep-jcl-return-codes-conditional-execution"

rm -rf "$TMP"
mkdir -p "$TMP"
unzip -o "$ZIP" -d "$TMP"

rm -rf "$LAB"
mkdir -p labs
cp -r "$TMP/32-multistep-jcl-return-codes-conditional-execution" "$LAB"
rm -rf "$TMP"

echo "===== LAB 32 INSTALADO ====="
find "$LAB" -type f | sort

echo "===== SECURITY CHECK ====="
grep -RInE \
'192\.168\.[0-9]{1,3}\.[0-9]{1,3}|10\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}|172\.(1[6-9]|2[0-9]|3[01])\.[0-9]{1,3}\.[0-9]{1,3}|([0-9A-Fa-f]{2}[:-]){5}[0-9A-Fa-f]{2}' \
"$LAB" || echo "OK - no private IP/MAC patterns found"

git status --short
git add "$LAB"
git commit -m "Add Lab 32 multi-step JCL conditional execution"
git push origin main

git status
git log -3 --oneline
git ls-remote origin refs/heads/main
